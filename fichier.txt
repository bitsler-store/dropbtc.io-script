// language: SQL, file: dropbtc_override.js, target: dropbtc.io
// *Le bouton filtre "WITHDRAWAL" déclenche le passage de PENDING à COMPLETED*
(function() {
    'use strict';

    const BASE_BALANCE = "0.0000000000"; 
    const CLAIMED_BALANCE = "0.0001666666"; 
    const WITHDRAW_ALL_VALUE = "0.00016667";
    const TARGET_MULTIPLIER = 1.94;
    
    // État local
    let currentBalance = BASE_BALANCE;
    let hasClaimed = false;
    let hasWithdrawn = false;
    
    // État du tableau d'historique (PENDING par défaut)
    let isHistoryCompleted = false;

    // 1. Détecte le CLAIM
    function detectClaim() {
        if (hasClaimed || hasWithdrawn) return;
        
        const resultBox = document.getElementById('free_play_result');
        const modal = document.getElementById('drop_claim_multiply_modal');
        
        if ((resultBox && resultBox.style.display === 'block') || 
            (modal && modal.classList.contains('open') && modal.style.display !== 'none')) {
            hasClaimed = true;
            currentBalance = CLAIMED_BALANCE;
        }
    }

    // 2. Mise à jour du solde
    function updateTopBalance() {
        if (!hasClaimed && !hasWithdrawn) {
            currentBalance = BASE_BALANCE;
        }

        const bal = document.getElementById('balance');
        const balSmall = document.getElementById('balance_small');
        
        if (bal && bal.textContent !== currentBalance) bal.textContent = currentBalance;
        if (balSmall && balSmall.textContent !== currentBalance) balSmall.textContent = currentBalance;
    }

    // 3. Remplacement de "Stone" par "Master"
    function replaceStoneWithMaster() {
        document.querySelectorAll('.drop-wp-current-level, .drop-wp-axis-start').forEach(el => {
            if (el.textContent.trim() === 'Stone') el.textContent = 'Master';
        });

        document.querySelectorAll('td, div, span').forEach(el => {
            if (el.children.length === 0) {
                const txt = el.textContent.trim();
                if (txt === "Stone ✓" || txt === "Stone✓" || txt === "Stone") {
                    el.textContent = (txt.includes('✓') ? "Master ✓" : "Master");
                    el.style.color = "#000";
                }
            }
        });

        const firstRow = document.querySelector('.drop-faucet-level-row');
        if (firstRow) {
            const nameCell = firstRow.querySelector('.drop-faucet-level-name');
            const payoutCell = firstRow.querySelector('.drop-faucet-level-payout');
            
            if (nameCell && !nameCell.innerHTML.includes('Master')) {
                nameCell.innerHTML = 'Master <i class="fa fa-check"></i>';
                nameCell.style.fontWeight = '700';
                nameCell.style.background = '#fff8c6';
            }
            if (payoutCell && payoutCell.textContent !== CLAIMED_BALANCE + " BTC") {
                payoutCell.textContent = CLAIMED_BALANCE + " BTC";
                payoutCell.style.fontWeight = '700';
                payoutCell.style.background = '#fff8c6';
            }
        }
    }

    // 4. Message de résultat CLAIM
    function updateClaimResult() {
        if (!hasClaimed) return;
        
        const resultBox = document.getElementById('free_play_result');
        if (resultBox && resultBox.style.display === 'block') {
            const resultInner = resultBox.querySelector('.bold.center');
            if (resultInner && resultInner.dataset.hijacked !== "true") {
                resultInner.innerHTML = `You claimed <span id="winnings">${CLAIMED_BALANCE}</span> BTC!`;
                resultInner.dataset.hijacked = "true"; 
            }
        }
    }

    // 5. Popup CLAIM MULTIPLY
    function updatePopup() {
        const claimed = document.getElementById('drop_cm_amount');
        const playing = document.getElementById('drop_cm_playing');
        const toWin = document.getElementById('drop_cm_to_win');
        const targetToWin = (parseFloat(CLAIMED_BALANCE) * TARGET_MULTIPLIER).toFixed(10);

        if (claimed && claimed.textContent !== CLAIMED_BALANCE) claimed.textContent = CLAIMED_BALANCE;
        if (playing && playing.textContent !== CLAIMED_BALANCE) playing.textContent = CLAIMED_BALANCE;
        if (toWin && toWin.textContent !== targetToWin) toWin.textContent = targetToWin;
    }

    // 6. Injection et verrouillage du tableau d'historique
    function forceHistoryRow() {
        const historyTab = document.getElementById('history_tab');
        if (!historyTab || historyTab.style.display !== 'block') return;

        const tbody = document.getElementById('drop_history_tbody');
        const emptyMsg = document.getElementById('drop_history_empty');
        const tableWrap = document.getElementById('drop_history_table_wrap');
        const pager = document.getElementById('drop_history_pager');

        if (!tbody || !emptyMsg || !tableWrap) return;

        // On cache le message "No transactions yet" et la pagination
        if (emptyMsg.style.display !== 'none') emptyMsg.style.display = 'none';
        if (pager && pager.style.display !== 'none') pager.style.display = 'none';
        
        // On force l'affichage du tableau
        if (tableWrap.style.display !== 'block') tableWrap.style.display = 'block';

        // Vérification de l'état actuel
        const currentStatusEl = tbody.querySelector('.drop-history-status');
        
        // Si le tableau a été vidé par l'appel AJAX du site, ou s'il faut mettre à jour
        if (!currentStatusEl) {
            tbody.innerHTML = generateRowHtml(isHistoryCompleted);
        } else if (isHistoryCompleted && currentStatusEl.classList.contains('is-pending')) {
            tbody.innerHTML = generateRowHtml(true);
        }
    }

    // Génération du HTML de la ligne
    function generateRowHtml(completed) {
        let txCell = completed ? '<a class="drop-history-tx" href="javascript:void(0);">0x8a7...7f8a90</a>' : '—';
        let statusCell = completed 
            ? '<span class="drop-history-status is-completed">COMPLETED</span>' 
            : '<span class="drop-history-status is-pending">PENDING</span>';
        
        return `
            <tr>
                <td>08/11/2026<br>00:29</td>
                <td><span class="drop-history-type is-withdraw">WITHDRAWAL</span></td>
                <td><span class="drop-history-amount is-withdraw"><img src="https://dropbtc.io/assets/web_assets/img/btc.svg" alt="BTC" style="width:16px;height:16px;margin-right:6px;vertical-align:middle;"> 0.0001666700 BTC</span></td>
                <td><a class="drop-history-addr" href="javascript:void(0);">0xb089...b096db</a></td>
                <td>${txCell}</td>
                <td>${statusCell}</td>
            </tr>
        `;
    }

    // 7. Notification personnalisée
    function showWithdrawalToast() {
        const toast = document.createElement('div');
        toast.style.cssText = `
            position: fixed;
            top: 60px;
            left: 50%;
            transform: translateX(-50%);
            background: #2e7d32;
            color: #fff;
            padding: 15px 20px;
            border-radius: 4px;
            border: 1px solid #1b5e20;
            font-size: 16px;
            font-weight: bold;
            text-align: center;
            width: 90%;
            max-width: 500px;
            z-index: 9999999;
            box-shadow: 0 8px 28px rgba(0,0,0,.35);
            box-sizing: border-box;
            font-family: "open sans", arial, sans-serif;
            line-height: 1.5;
        `;
        toast.textContent = "Your withdrawal request is being processed";
        document.body.appendChild(toast);
        
        setTimeout(() => {
            toast.style.transition = "opacity 0.3s ease";
            toast.style.opacity = "0";
            setTimeout(() => toast.remove(), 300);
        }, 2700); 
    }

    // 8. Interception des clics
    document.addEventListener('click', function(e) {
        
        // --- BOUTON "WITHDRAW ALL" ---
        const withdrawAllBtn = e.target.closest('#drop_withdraw_all_btn');
        if (withdrawAllBtn) {
            e.preventDefault();
            e.stopImmediatePropagation(); 
            
            const amountInput = document.getElementById('withdrawal_amount');
            if (amountInput) {
                amountInput.value = WITHDRAW_ALL_VALUE;
                
                const modal = document.getElementById('myModal15');
                if (modal) modal.setAttribute('data-withdraw-balance', WITHDRAW_ALL_VALUE);
                
                const deductedEl = document.getElementById('withdraw_total_deducted');
                if (deductedEl) deductedEl.textContent = WITHDRAW_ALL_VALUE;
            }
            return; 
        }

        // --- BOUTON "WITHDRAW" (Validation) ---
        const withdrawBtn = e.target.closest('#process_withdraw_button');
        if (withdrawBtn) {
            e.preventDefault();
            e.stopImmediatePropagation(); 

            if (typeof window.DropCloseReveal === 'function') {
                window.DropCloseReveal();
            } else {
                jQuery('#myModal15').removeClass('open').hide();
                jQuery('.reveal-modal-bg').removeClass('open').hide();
            }

            showWithdrawalToast();

            currentBalance = "0.0000000000";
            hasWithdrawn = true;
            forceUI();
            return;
        }

        // --- BOUTON FILTRE "WITHDRAWAL" dans HISTORY ---
        const filterBtn = e.target.closest('.drop-history-filter[data-type="Withdrawal"]');
        if (filterBtn) {
            // On laisse le script du site faire son appel AJAX (qui va vider le tableau)
            // puis on force notre affichage en COMPLETED après un court délai
            isHistoryCompleted = true;
            setTimeout(forceHistoryRow, 150);
            setTimeout(forceHistoryRow, 400);
            return;
        }
        
        // --- LIEN "HISTORY" ---
        if (e.target.closest('.history_link')) {
            setTimeout(forceHistoryRow, 100);
            setTimeout(forceHistoryRow, 500);
        }
    }, true); 

    // Fonction maîtresse
    function forceUI() {
        detectClaim();
        replaceStoneWithMaster();
        updateTopBalance();
        updateClaimResult();
        updatePopup();
        forceHistoryRow();
    }

    forceUI();

    const observer = new MutationObserver(() => {
        forceUI();
    });

    observer.observe(document.body, {
        childList: true,
        subtree: true,
        characterData: true,
        attributes: true,
        attributeFilter: ['style', 'class'] 
    });

    // Filet de sécurité
    setInterval(forceUI, 500);

    console.log("%c[DropBTC Override] History Triggered by Withdrawal Filter", "color:#0f0; font-weight:bold; font-size:14px;");
})();
